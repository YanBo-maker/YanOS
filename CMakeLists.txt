cmake_minimum_required(VERSION 3.21)
project(YanOS LANGUAGES C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(MSVC)
    add_compile_options(/W4 /WX)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()

option(YAN_ENABLE_SANITIZERS "Check address safety and undefined behavior" OFF)
if(YAN_ENABLE_SANITIZERS)
    if(UNIX AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer
                            -fno-sanitize-recover=all)
        add_link_options(-fsanitize=address,undefined -fno-sanitize-recover=all)
    else()
        message(FATAL_ERROR "Sanitizer builds require GCC or Clang on Unix")
    endif()
endif()

add_library(yan_machine STATIC src/ram.c src/bus.c src/machine.c src/cpu.c)
target_include_directories(yan_machine PUBLIC include)

add_executable(memory_demo examples/memory_demo.c)
target_link_libraries(memory_demo PRIVATE yan_machine)

add_executable(fetch_demo examples/fetch_demo.c)
target_link_libraries(fetch_demo PRIVATE yan_machine)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_ram tests/test_ram.c)
    target_link_libraries(test_ram PRIVATE yan_machine)
    add_test(NAME ram COMMAND test_ram)
    add_executable(test_bus tests/test_bus.c)
    target_link_libraries(test_bus PRIVATE yan_machine)
    add_test(NAME bus COMMAND test_bus)
    add_executable(test_machine tests/test_machine.c)
    target_link_libraries(test_machine PRIVATE yan_machine)
    add_test(NAME machine COMMAND test_machine)
    add_test(NAME memory_demo COMMAND memory_demo)
    add_executable(test_cpu tests/test_cpu.c)
    target_link_libraries(test_cpu PRIVATE yan_machine)
    add_test(NAME cpu COMMAND test_cpu)
    add_executable(test_fetch tests/test_fetch.c)
    target_link_libraries(test_fetch PRIVATE yan_machine)
    add_test(NAME fetch COMMAND test_fetch)
    add_test(NAME fetch_demo COMMAND fetch_demo)
    set_tests_properties(ram bus machine memory_demo cpu fetch fetch_demo PROPERTIES TIMEOUT 10)
endif()
